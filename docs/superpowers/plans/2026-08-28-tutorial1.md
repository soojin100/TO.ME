# 튜토리얼 1 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 기획서 `튜토리얼1.pdf` p2~p16의 흐름(시작 화면 분기 → 강아지 소환 → 이름 입력 → 나레이션/스탠딩 연출 → 무공격 에너미 → 제약 걸린 튜토리얼 전투)을 수치·텍스트 하드코딩 없이 구현한다.

**Architecture:** 대사 **밖** 연출은 `TutorialSequenceSO`의 `[SerializeReference]` 스텝 배열로 데이터화하고 `TutorialSequenceRunner`가 코루틴으로 순차 실행한다. 대사 **중** 연출(스탠딩 교체·흔들림)은 `dialogue` 시트의 `standing`/`effect` 컬럼으로 줄 단위 지정한다. ScriptableObject는 씬 오브젝트를 참조할 수 없으므로 스텝은 문자열 키로 대상을 가리키고, 씬의 `TutorialTarget` 컴포넌트가 키를 부여한다.

**Tech Stack:** Unity 6000.4.0f1, C#, NUnit(EditMode), TextMeshPro, 기존 `TOME.*` 네임스페이스 규약

**설계 문서:** `docs/superpowers/specs/2026-08-28-tutorial1-design.md`

이 계획의 Task 1~11은 스크립트·테스트만 다루므로 worktree에서 진행한다.

Map_Room 진입 시 `TutorialSequenceRunner`가 `SeenIntro`를 보고 스스로 시작한다.
검증도 Map_Room 씬에서 직접 Play 한다.

브리지가 응답하지 않으면 Unity에서 `Window > MCP For Unity`로 연결 상태를 확인한다.
Unity 없이 작업하는 동안에는 각 Task의 EditMode 테스트가 컴파일 검증을 대신한다.

```csharp
using NUnit.Framework;
using UnityEngine;
using TOME.Utils;

namespace TOME.Tests.EditMode
{
    public class DialogueColumnTests
    {
        static TextAsset Csv(string body) => new TextAsset(body);

        [Test]
        public void Load_ReadsStandingAndEffectColumns()
        {
            var csv = Csv(
                "id,speaker,text,next,chapter,speakerSprite,trigger,standing,effect\n" +
                "a,강아지,화났다,b,tut1,,,rage,shake\n");

            var table = CsvImporter.LoadDialogue(csv);

            Assert.AreEqual("rage",  table["a"].standing);
            Assert.AreEqual("shake", table["a"].effect);
        }

        [Test]
        public void Load_MissingColumns_YieldEmptyStrings()
        {
            // 기존 CSV(컬럼 없음)도 그대로 읽혀야 한다
            var csv = Csv(
                "id,speaker,text,next,chapter,speakerSprite,trigger\n" +
                "a,강아지,안녕,b,chapter1,,\n");

            var table = CsvImporter.LoadDialogue(csv);

            Assert.AreEqual(string.Empty, table["a"].standing);
            Assert.AreEqual(string.Empty, table["a"].effect);
            Assert.AreEqual("안녕", table["a"].text);
        }
    }
}
```

```csharp
            // 버튼을 숨기지 못한 경로(직접 호출 등)도 여기서 차단한다.

            return stage == null || stage.allowInventory;
        }
```

```csharp
using System.Collections.Generic;
using UnityEngine;

namespace TOME.Tutorial
{
    /// <summary>스텝 실행에 필요한 씬 정보. 러너가 만들어 각 스텝에 넘긴다.</summary>
    public class TutorialContext
    {
        readonly Dictionary<string, GameObject> _targets;

        public TutorialContext(Dictionary<string, GameObject> targets)
        {
            _targets = targets ?? new Dictionary<string, GameObject>();
        }

        /// <summary>키에 해당하는 씬 오브젝트. 없으면 경고 후 null.</summary>
        public GameObject Find(string key)
        {
            if (string.IsNullOrEmpty(key)) return null;
            if (_targets.TryGetValue(key, out var go)) return go;
            Debug.LogWarning($"[Tutorial] TutorialTarget '{key}' 을(를) 씬에서 찾지 못했습니다.");
            return null;
        }

        /// <summary>키에 해당하는 오브젝트의 컴포넌트. 비활성 오브젝트도 찾는다.</summary>
        public T FindComponent<T>(string key) where T : Component
        {
            var go = Find(key);
            if (go == null) return null;
            var c = go.GetComponent<T>();
            if (c == null)
                Debug.LogWarning($"[Tutorial] '{key}' 에 {typeof(T).Name} 컴포넌트가 없습니다.");
            return c;
        }
    }
}
```

```csharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using TOME.Core;

namespace TOME.Tutorial
{
    /// <summary>씬에 배치해 TutorialSequenceSO의 스텝을 순차 실행한다.
    /// 이미 튜토리얼을 본 세이브(SeenIntro)에서는 아무것도 하지 않는다.</summary>
    public class TutorialSequenceRunner : MonoBehaviour
    {
        public static TutorialSequenceRunner I { get; private set; }

        [SerializeField] TutorialSequenceSO sequence;
        [Tooltip("체크하면 SeenIntro가 true인 세이브에서 시퀀스를 건너뛴다.")]
        [SerializeField] bool skipWhenIntroSeen = true;
        [Tooltip("시퀀스 완료 시 SaveSystemManager.MarkIntroSeen()을 호출한다.")]
        [SerializeField] bool markIntroSeenOnComplete = true;

        public bool IsRunning { get; private set; }

        void Awake() { I = this; }
        void OnDestroy() { if (I == this) I = null; }

        IEnumerator Start()
        {
            if (sequence == null || sequence.steps == null || sequence.steps.Count == 0) yield break;
            if (skipWhenIntroSeen && SaveSystemManager.I != null && SaveSystemManager.I.SeenIntro)
                yield break;

            var ctx = new TutorialContext(BuildTargetTable());
            IsRunning = true;

            foreach (var step in sequence.steps)
            {
                if (step == null) continue;
                yield return step.Execute(ctx);
            }

            IsRunning = false;
            if (markIntroSeenOnComplete) SaveSystemManager.I?.MarkIntroSeen();
        }

        // 비활성 오브젝트도 포함한다 — 이름 판자·스탠딩 UI는 시작 시 꺼져 있다.
        Dictionary<string, GameObject> BuildTargetTable()
        {
            var table = new Dictionary<string, GameObject>(16);
            var targets = FindObjectsByType<TutorialTarget>(
                FindObjectsInactive.Include, FindObjectsSortMode.None);

            foreach (var t in targets)
            {
                if (t == null || string.IsNullOrEmpty(t.Key)) continue;
                if (table.ContainsKey(t.Key))
                {
                    Debug.LogWarning($"[Tutorial] 중복 TutorialTarget 키 '{t.Key}' — 먼저 찾은 것을 씁니다.");
                    continue;
                }
                table[t.Key] = t.gameObject;
            }
            return table;
        }
    }
}
```

```csharp
using System.Collections.Generic;
using UnityEngine;

namespace TOME.Tutorial
{
    /// <summary>키프레임 스케일 보간. 기획서의 "0 → 1.3 → 0.9 → 1.08 → 1.0" 같은
    /// 펀치 등장 연출을 배열 데이터로만 표현하기 위한 순수 함수 모음.
    /// keys가 n개면 구간은 n-1개이며 durations[i]가 i번째 구간의 소요 시간이다.
    /// durations가 짧으면 마지막 값을 반복 사용한다.</summary>
    public static class ScaleSequence
    {
        /// <summary>구간 i의 소요 시간. durations가 비었으면 0.</summary>
        public static float SegmentDuration(IReadOnlyList<float> durations, int i)
        {
            if (durations == null || durations.Count == 0) return 0f;
            return durations[Mathf.Min(i, durations.Count - 1)];
        }

        /// <summary>전체 재생 시간. 키 사이 구간(n-1개)만 합산한다.</summary>
        public static float TotalDuration(IReadOnlyList<float> keys, IReadOnlyList<float> durations)
        {
            if (keys == null || keys.Count < 2) return 0f;
            float total = 0f;
            for (int i = 0; i < keys.Count - 1; i++) total += SegmentDuration(durations, i);
            return total;
        }

        /// <summary>시각 t에서의 스케일. keys가 비었으면 1(원본 크기)을 돌려준다.</summary>
        public static float Evaluate(IReadOnlyList<float> keys, IReadOnlyList<float> durations, float t)
        {
            if (keys == null || keys.Count == 0) return 1f;
            if (keys.Count == 1) return keys[0];
            if (t <= 0f) return keys[0];

            float elapsed = 0f;
            for (int i = 0; i < keys.Count - 1; i++)
            {
                float dur = SegmentDuration(durations, i);
                if (dur <= 0f)
                {
                    // 시간 0 구간은 즉시 통과 — 다음 키로 점프.
                    elapsed += dur;
                    continue;
                }
                if (t < elapsed + dur)
                    return Mathf.Lerp(keys[i], keys[i + 1], (t - elapsed) / dur);
                elapsed += dur;
            }
            return keys[keys.Count - 1];
        }
    }
}
```

```csharp
        /// <summary>공백을 뗀 길이가 [min, max] 안인지. UI 없이 검증만 하고 싶을 때 쓴다.</summary>
        public static bool IsLengthValid(string raw, int min, int max)
        {
            string trimmed = (raw ?? string.Empty).Trim();
            return trimmed.Length >= min && trimmed.Length <= max;
        }

        void Awake()
        {
            if (board) board.anchoredPosition = hiddenPosition;
            if (invalidHint) invalidHint.SetActive(false);
            if (confirmButton) confirmButton.onClick.AddListener(OnConfirm);
            if (inputField) inputField.onValueChanged.AddListener(OnValueChanged);
        }

        void OnDestroy()
        {
            if (confirmButton) confirmButton.onClick.RemoveListener(OnConfirm);
            if (inputField) inputField.onValueChanged.RemoveListener(OnValueChanged);
        }

        /// <summary>판자를 내려 입력을 받고, 확정되면 저장 후 아래로 퇴장한다.
        /// 스텝이 이 코루틴을 yield 해 완료를 기다린다.</summary>
        public IEnumerator RunRoutine(int minLength, int maxLength)
        {
            _min = Mathf.Max(1, minLength);
            _max = Mathf.Max(_min, maxLength);
            _confirmed = false;

            gameObject.SetActive(true);
            if (inputField)
            {
                inputField.characterLimit = _max;   // 최대 길이는 입력 단계에서 막는다
                inputField.text = "";
            }
            RefreshConfirmState();

            yield return SlideRoutine(hiddenPosition, shownPosition, slideInDuration);

            if (inputField) inputField.ActivateInputField();
            while (!_confirmed) yield return null;

            if (inputField) inputField.DeactivateInputField();
            yield return SlideRoutine(shownPosition, exitPosition, slideOutDuration);
            gameObject.SetActive(false);
        }

        void OnValueChanged(string _) => RefreshConfirmState();

        void RefreshConfirmState()
        {
            bool ok = IsLengthValid(inputField ? inputField.text : null, _min, _max);
            if (confirmButton) confirmButton.interactable = ok;
            if (invalidHint) invalidHint.SetActive(!ok);
        }

        void OnConfirm()
        {
            string entered = inputField ? inputField.text : null;
            if (!IsLengthValid(entered, _min, _max)) return;   // 버튼을 못 막은 경로도 차단
            SaveSystemManager.I?.SetPlayerName(entered.Trim());
            _confirmed = true;
        }

        IEnumerator SlideRoutine(Vector2 from, Vector2 to, float duration)
        {
            if (board == null) yield break;
            if (duration <= 0f) { board.anchoredPosition = to; yield break; }

            float t = 0f;
            while (t < duration)
            {
                t += Time.unscaledDeltaTime;
                board.anchoredPosition = Vector2.LerpUnclamped(
                    from, to, slideCurve.Evaluate(Mathf.Clamp01(t / duration)));
                yield return null;
            }
            board.anchoredPosition = to;
        }
    }
}
```

```shaderlab
Shader "TOME/RoundedSpotlight"
{
    Properties
    {
        _Color      ("Dim Color", Color) = (0,0,0,1)
        _Center     ("Spotlight Center (world xy)", Vector) = (0,0,0,0)
        _HalfSize   ("Spotlight Half Size (world xy)", Vector) = (0,0,0,0)
        _Radius     ("Corner Radius (world)", Float) = 0.5
        _Softness   ("Edge Softness (world)", Float) = 0.15
        _DimAmount  ("Dim Amount", Range(0,1)) = 0.55
    }
    SubShader
    {
        Tags { "Queue"="Transparent" "RenderType"="Transparent" "IgnoreProjector"="True" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes { float4 positionOS : POSITION; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 worldXY : TEXCOORD0; };

            float4 _Color;
            float4 _Center;
            float4 _HalfSize;
            float  _Radius;
            float  _Softness;
            float  _DimAmount;

            Varyings vert (Attributes v)
            {
                Varyings o;
                float3 wp = TransformObjectToWorld(v.positionOS.xyz);
                o.positionCS = TransformWorldToHClip(wp);
                o.worldXY = wp.xy;
                return o;
            }

            // 둥근 사각형 SDF. 0보다 작으면 내부.
            float RoundedBoxSDF(float2 p, float2 halfSize, float r)
            {
                float2 q = abs(p) - (halfSize - r);
                return length(max(q, 0.0)) + min(max(q.x, q.y), 0.0) - r;
            }

            half4 frag (Varyings i) : SV_Target
            {
                // HalfSize가 0이면 스포트라이트 없음 → 화면 전체 균일 암전.
                float cover = 1.0;
                if (_HalfSize.x > 0.0001 && _HalfSize.y > 0.0001)
                {
                    float r = min(_Radius, min(_HalfSize.x, _HalfSize.y));
                    float d = RoundedBoxSDF(i.worldXY - _Center.xy, _HalfSize.xy, r);
                    // 내부(d<0)는 0, 외부는 1. _Softness 폭으로 부드럽게.
                    cover = smoothstep(0.0, max(_Softness, 0.0001), d);
                }
                half4 c = _Color;
                c.a *= cover * _DimAmount;
                return c;
            }
            ENDHLSL
        }
    }
}
```

```csharp
using System.Collections;
using UnityEngine;

namespace TOME.Tutorial
{
    /// <summary>화면을 어둡게 덮고, 지정 오브젝트 영역만 둥근 모서리 사각형으로 뚫어 남긴다.
    /// 오브젝트별 색을 만지지 않는 이유: 기획서 p4 예시가 대상 *주변 영역*(뒤 벽·바닥 포함)을
    /// 통째로 밝게 남기기 때문. 오브젝트 단위로 낮추면 대상만 밝고 뒤 바닥이 어두워져 오려낸 것처럼 보인다.
    /// 대상 없이 호출하면 스포트라이트 없는 균일 암전이 되어 나레이션 모드가 재사용한다.</summary>
    [RequireComponent(typeof(MeshRenderer), typeof(MeshFilter))]
    public class SpotlightDimmer : MonoBehaviour
    {
        [Tooltip("오버레이가 따라갈 카메라. 비우면 Camera.main.")]
        [SerializeField] Camera targetCamera;
        [Tooltip("RoundedSpotlight 셰이더를 쓰는 머티리얼.")]
        [SerializeField] Material overlayMaterial;
        [Tooltip("오버레이 정렬 순서. 맵 오브젝트보다 앞, 대화 UI보다 뒤.")]
        [SerializeField] int sortingOrder = 500;
        [Tooltip("스포트라이트 모서리 부드러움(월드 단위).")]
        [SerializeField] float edgeSoftness = 0.15f;

        static readonly int CenterId    = Shader.PropertyToID("_Center");
        static readonly int HalfSizeId  = Shader.PropertyToID("_HalfSize");
        static readonly int RadiusId    = Shader.PropertyToID("_Radius");
        static readonly int SoftnessId  = Shader.PropertyToID("_Softness");
        static readonly int DimAmountId = Shader.PropertyToID("_DimAmount");

        MeshRenderer _renderer;
        Material     _instance;   // 공유 머티리얼 오염 방지용 인스턴스
        float        _dimAmount;

        public bool IsDimmed => _dimAmount > 0.0001f;

        void Awake()
        {
            if (targetCamera == null) targetCamera = Camera.main;
            _renderer = GetComponent<MeshRenderer>();
            _renderer.sortingOrder = sortingOrder;

            if (overlayMaterial != null)
            {
                _instance = new Material(overlayMaterial);
                _renderer.material = _instance;
            }
            SetDim(0f);
            SetSpotlight(Vector2.zero, Vector2.zero, 0f);
            _renderer.enabled = false;
        }

        void OnDestroy() { if (_instance != null) Destroy(_instance); }

        // 카메라가 좌우로 움직여도(ScreenNavigator) 오버레이가 화면을 계속 덮게 한다.
        void LateUpdate()
        {
            if (!_renderer.enabled || targetCamera == null || !targetCamera.orthographic) return;
            FitToCamera();
        }

        void FitToCamera()
        {
            var camT = targetCamera.transform;
            float h = targetCamera.orthographicSize * 2f;
            float w = h * targetCamera.aspect;
            // 회전/오차 여유로 살짝 크게 잡는다.
            transform.position   = new Vector3(camT.position.x, camT.position.y, camT.position.z + 1f);
            transform.localScale = new Vector3(w * 1.05f, h * 1.05f, 1f);
        }

        /// <summary>target을 감싸는 둥근 사각형만 남기고 암전한다.
        /// target이 null이면 스포트라이트 없이 화면 전체를 균일하게 어둡게 한다.</summary>
        public IEnumerator DimRoutine(GameObject target, float padding, float cornerRadius,
                                      float dimAmount, float duration)
        {
            _renderer.enabled = true;
            FitToCamera();

            if (target != null && TryGetBounds(target, out Bounds b))
            {
                var half = new Vector2(b.extents.x + padding, b.extents.y + padding);
                SetSpotlight(b.center, half, cornerRadius);
            }
            else
            {
                SetSpotlight(Vector2.zero, Vector2.zero, 0f);
            }

            yield return FadeRoutine(_dimAmount, Mathf.Clamp01(dimAmount), duration);
        }

        /// <summary>암전을 걷어낸다.</summary>
        public IEnumerator UndimRoutine(float duration)
        {
            if (!IsDimmed) { _renderer.enabled = false; yield break; }
            yield return FadeRoutine(_dimAmount, 0f, duration);
            _renderer.enabled = false;
        }

        // 대상과 그 자식들의 렌더러를 모두 감싸는 바운즈. 여러 스프라이트로 된 오브젝트 대응.
        static bool TryGetBounds(GameObject go, out Bounds bounds)
        {
            bounds = default;
            var renderers = go.GetComponentsInChildren<Renderer>(true);
            bool any = false;
            foreach (var r in renderers)
            {
                if (r == null) continue;
                if (!any) { bounds = r.bounds; any = true; }
                else bounds.Encapsulate(r.bounds);
            }
            return any;
        }

        void SetSpotlight(Vector2 center, Vector2 halfSize, float radius)
        {
            if (_instance == null) return;
            _instance.SetVector(CenterId,   new Vector4(center.x, center.y, 0f, 0f));
            _instance.SetVector(HalfSizeId, new Vector4(halfSize.x, halfSize.y, 0f, 0f));
            _instance.SetFloat(RadiusId,    radius);
            _instance.SetFloat(SoftnessId,  edgeSoftness);
        }

        void SetDim(float amount)
        {
            _dimAmount = amount;
            if (_instance != null) _instance.SetFloat(DimAmountId, amount);
        }

        IEnumerator FadeRoutine(float from, float to, float duration)
        {
            if (duration <= 0f) { SetDim(to); yield break; }
            float t = 0f;
            while (t < duration)
            {
                t += Time.unscaledDeltaTime;
                SetDim(Mathf.Lerp(from, to, Mathf.Clamp01(t / duration)));
                yield return null;
            }
            SetDim(to);
        }
    }
}
```

```csharp
using UnityEngine;

namespace TOME.Tutorial
{
    /// <summary>나레이션 구간 동안 화면을 균일하게 어둡게 하고 맵 오브젝트의 콜라이더를 꺼
    /// 대화창·스킵 버튼만 동작하게 한다.
    /// SetActive(false)를 쓰지 않는 이유: MapBusyVisibility가 이미 대화 중 SetActive를 토글하고 있어
    /// 두 시스템이 같은 오브젝트를 켜고 끄면 복원 상태가 꼬인다. 암전·콜라이더만 만지면 겹쳐도 안전하다.</summary>
    public class NarrationModeController : MonoBehaviour
    {
        [Tooltip("암전을 담당할 SpotlightDimmer. 스포트라이트 없이(=균일) 사용한다.")]
        [SerializeField] SpotlightDimmer dimmer;
        [Tooltip("콜라이더를 끌 범위. 보통 맵 배경 루트(예: Room_Background).")]
        [SerializeField] Transform contentRoot;
        [Tooltip("입력을 살려 둘 오브젝트(자식 포함).")]
        [SerializeField] GameObject[] keepInteractive;
        [Range(0f, 1f)]
        [Tooltip("나레이션 중 화면 어둡기. 0=그대로, 1=완전 암전.")]
        [SerializeField] float dimAmount = 0.55f;
        [SerializeField] float fadeDuration = 0.35f;

        readonly List<Collider2D> _disabled = new(32);
        bool _active;

        public bool IsActive => _active;

        /// <summary>나레이션 모드 진입. 스텝이 이 코루틴을 yield 해 완료를 기다린다.</summary>
        public IEnumerator EnterRoutine()
        {
            if (_active) yield break;
            _active = true;

            // 콜라이더를 먼저 꺼서 페이드 도중 클릭이 새지 않게 한다.
            CollectAndDisableColliders();

            if (dimmer != null)
                yield return dimmer.DimRoutine(null, 0f, 0f, dimAmount, fadeDuration);
        }

        /// <summary>나레이션 모드 종료. 암전과 콜라이더를 원복한다.</summary>
        public IEnumerator ExitRoutine()
        {
            if (!_active) yield break;
            if (dimmer != null) yield return dimmer.UndimRoutine(fadeDuration);
            Restore();
        }

        void OnDisable() { if (_active) Restore(); }

        void CollectAndDisableColliders()
        {
            _disabled.Clear();
            var root = contentRoot != null ? contentRoot : transform;

            foreach (var c in root.GetComponentsInChildren<Collider2D>(true))
            {
                if (c == null || !c.enabled || IsKept(c.transform)) continue;
                c.enabled = false;
                _disabled.Add(c);
            }
        }

        bool IsKept(Transform t)
        {
            if (keepInteractive == null) return false;
            foreach (var go in keepInteractive)
            {
                if (go == null) continue;
                if (t == go.transform || t.IsChildOf(go.transform)) return true;
            }
            return false;
        }

        void Restore()
        {
            foreach (var c in _disabled)
                if (c != null) c.enabled = true;
            _disabled.Clear();
            _active = false;
        }
    }
}
```

```csharp
        {
            if (!_subscribed) return;
            if (DialogueManager.I != null)
            {
                DialogueManager.I.OnLine -= OnLine;
                DialogueManager.I.OnEnd  -= OnDialogueEnd;
            }
            _subscribed = false;
        }

        void OnLine(DialogueEntry e)
        {
            // 빈칸이면 직전 스탠딩을 그대로 유지한다(줄마다 다시 적지 않아도 되게).
            if (!string.IsNullOrWhiteSpace(e.standing)) ApplyStanding(e.standing.Trim());
            ApplyEffect(e.effect);
        }

        void OnDialogueEnd()
        {
            SetShaking(false);
            Hide();
        }

        void ApplyStanding(string key)
        {
            if (string.Equals(key, HideKey, System.StringComparison.OrdinalIgnoreCase)) { Hide(); return; }

            var sprite = FindSprite(key);
            if (sprite == null)
            {
                Debug.LogWarning($"[Tutorial] 스탠딩 키 '{key}' 에 해당하는 스프라이트가 없습니다.");
                return;
            }
            if (image == null) return;
            image.sprite = sprite;
            image.enabled = true;
            if (!gameObject.activeSelf) gameObject.SetActive(true);
        }

        void ApplyEffect(string effect)
        {
            bool shake = !string.IsNullOrWhiteSpace(effect)
                      && string.Equals(effect.Trim(), ShakeEffect, System.StringComparison.OrdinalIgnoreCase);
            SetShaking(shake);
        }

        Sprite FindSprite(string key)
        {
            if (standings == null) return null;
            foreach (var s in standings)
                if (string.Equals(s.key, key, System.StringComparison.OrdinalIgnoreCase)) return s.sprite;
            return null;
        }

        void Hide()
        {
            SetShaking(false);
            if (image) image.enabled = false;
        }

        void SetShaking(bool on)
        {
            if (on)
            {
                if (_shakeCo == null && shakeTarget != null) _shakeCo = StartCoroutine(ShakeRoutine());
                return;
            }
            if (_shakeCo != null) { StopCoroutine(_shakeCo); _shakeCo = null; }
            if (shakeTarget != null) shakeTarget.anchoredPosition = _basePosition;
        }

        IEnumerator ShakeRoutine()
        {
            var wait = new WaitForSecondsRealtime(Mathf.Max(0.01f, shakeInterval));
            while (true)
            {
                float min = Mathf.Min(shakePixelRange.x, shakePixelRange.y);
                float max = Mathf.Max(shakePixelRange.x, shakePixelRange.y);
                // 축마다 독립적으로 부호와 거리를 뽑는다 → "위아래 좌우로" 흔들림.
                float dx = Random.Range(min, max) * (Random.value < 0.5f ? -1f : 1f);
                float dy = Random.Range(min, max) * (Random.value < 0.5f ? -1f : 1f);
                shakeTarget.anchoredPosition = _basePosition + new Vector2(dx, dy);
                yield return wait;
            }
        }
    }
}
```

```csharp
using System.Collections;
using UnityEngine;

namespace TOME.Tutorial
{
    /// <summary>기획서 p5: 클릭 시 클라이밍 애니 + 스케일 0→1.3→0.9→1.08→1.0 으로 툭 튀어나온 뒤,
    /// 강아지 집 중앙에서 화면 중앙으로 걸어간다(걷기 애니). 스케일 1.0 = 배치된 원본 크기.</summary>
    [System.Serializable]
    public class SummonCharacterStep : TutorialStep
    {
        [Tooltip("등장시킬 캐릭터의 TutorialTarget 키.")]
        public string characterKey = "dog";
        [Tooltip("등장 시작 위치의 TutorialTarget 키(강아지 집 중앙).")]
        public string spawnPointKey = "dogHouseCenter";
        [Tooltip("걸어갈 목적지의 TutorialTarget 키(화면 중앙).")]
        public string moveTargetKey = "roomCenter";

        [Header("등장 스케일 (1.0 = 배치된 원본 크기)")]
        public float[] scaleKeys      = { 0f, 1.3f, 0.9f, 1.08f, 1.0f };
        [Tooltip("키 사이 각 구간의 소요 시간(초). 키가 n개면 n-1개.")]
        public float[] scaleDurations = { 0.12f, 0.08f, 0.06f, 0.06f };

        [Header("애니메이션")]
        [Tooltip("등장 시 재생할 CharacterWander 등록 애니 ID.")]
        public string appearAnimationId = "climbing";
        [Tooltip("이동 중 켜 둘 Animator bool 파라미터. CharacterWander의 walkParam과 같은 이름.")]
        public string walkParam = "walk";

        [Header("이동")]
        [Min(0f)] public float moveSpeed = 2f;

        public override IEnumerator Execute(TutorialContext ctx)
        {
            var character = ctx.Find(characterKey);
            if (character == null) yield break;

            var spawn = ctx.Find(spawnPointKey);
            if (spawn != null) character.transform.position = spawn.transform.position;

            character.SetActive(true);

            var animator = character.GetComponentInChildren<Animator>();
            var wander   = character.GetComponent<TOME.Gameplay.CharacterWander>();

            // 등장 애니는 CharacterWander의 등록 목록을 통해 재생한다(파라미터 이름 중복 관리 회피).
            if (wander != null && !string.IsNullOrEmpty(appearAnimationId))
                wander.PlayAnimation(appearAnimationId);

            yield return PunchScaleRoutine(character.transform);

            var moveTarget = ctx.Find(moveTargetKey);
            if (moveTarget != null)
                yield return WalkRoutine(character.transform, moveTarget.transform.position, animator);
        }

        // 배치된 원본 스케일을 1.0 기준으로 삼아 키프레임 배율을 곱한다.
        IEnumerator PunchScaleRoutine(Transform t)
        {
            Vector3 baseScale = t.localScale;
            float total = ScaleSequence.TotalDuration(scaleKeys, scaleDurations);
            if (total <= 0f)
            {
                t.localScale = baseScale;
                yield break;
            }

            float elapsed = 0f;
            while (elapsed < total)
            {
                elapsed += Time.unscaledDeltaTime;
                float k = ScaleSequence.Evaluate(scaleKeys, scaleDurations, elapsed);
                t.localScale = baseScale * k;
                yield return null;
            }
            t.localScale = baseScale;
        }

        IEnumerator WalkRoutine(Transform t, Vector3 destination, Animator animator)
        {
            destination.z = t.position.z;

            // 진행 방향으로 좌우 반전 — CharacterWander와 같은 규약(부호 반전).
            float dx = destination.x - t.position.x;
            if (Mathf.Abs(dx) > 0.01f)
            {
                Vector3 s = t.localScale;
                t.localScale = new Vector3(Mathf.Abs(s.x) * -Mathf.Sign(dx), s.y, s.z);
            }

            if (animator != null && !string.IsNullOrEmpty(walkParam)) animator.SetBool(walkParam, true);

            while (Vector3.Distance(t.position, destination) > 0.05f)
            {
                t.position = Vector3.MoveTowards(
                    t.position, destination, moveSpeed * Time.unscaledDeltaTime);
                yield return null;
            }
            t.position = destination;

            if (animator != null && !string.IsNullOrEmpty(walkParam)) animator.SetBool(walkParam, false);
        }
    }
}
```

```csharp
namespace TOME.Tutorial
```

```csharp
    /// <summary>대사를 재생하고 끝날 때까지 기다린다. 나레이션 모드를 함께 켜고 끌 수 있다.</summary>
    [System.Serializable]
    public class PlayDialogueStep : TutorialStep
    {
        [Tooltip("dialogue 시트의 시작 ID. 예: tut1_nar_01")]
        public string startId;
        [Tooltip("체크하면 대사 시작 전 나레이션 모드로 들어가고 종료 후 나온다(기획서 p7).")]
        public bool useNarrationMode = true;
        [Tooltip("NarrationModeController 컴포넌트를 가진 오브젝트의 TutorialTarget 키.")]
        public string narrationKey = "narration";

        public override IEnumerator Execute(TutorialContext ctx)
        {
            if (string.IsNullOrEmpty(startId)) yield break;

            NarrationModeController narration = null;
            if (useNarrationMode)
            {
                narration = ctx.FindComponent<NarrationModeController>(narrationKey);
                if (narration != null) yield return narration.EnterRoutine();
            }

            if (DialogueManager.I != null && DialogueManager.I.TryPlay(startId))
            {
                // TryPlay는 시작만 알린다. IsPlaying이 내려갈 때까지 기다린다.
                while (DialogueManager.I.IsPlaying) yield return null;
            }
            else
            {
                Debug.Log($"[Tutorial] 대사 '{startId}' 재생 안 됨(이미 봤거나 표에 없음). 다음 스텝으로 진행.");
            }

            if (narration != null) yield return narration.ExitRoutine();
        }
    }
```
